## Global Constraints
- 시각은 모두 KST(+09:00). 현재 시각은 `now_kst()` 하나로만 만든다. 시각 컬럼은 모두 `KstDateTime`.
- 시간대 없는(naive) datetime은 저장하지 않는다 — 저장 시 `ValueError`.
- Enum은 대문자 문자열(`StrEnum`)로 정의하고 DB에는 문자열로 저장한다.
- 금액은 정수(원 단위).
- 비밀번호: PBKDF2-HMAC-SHA512, 200,000회, salt 16바이트, 형식 `pbkdf2_sha512$200000$<salt base64>$<hash base64>`, 비교는 `hmac.compare_digest`.
- 설정은 `get_settings()`로만 읽는다. `os.environ` 직접 읽기 금지.
- 파일 하나 200줄 이하(테스트 제외). 함수 인자·반환 타입 힌트 필수. `ruff check` 통과.
- 사용자에게 보이는 메시지는 한국어.
- 시드 비밀번호는 시연용 테스트 값이며 `app/seed.py`에만 둔다. 실제 비밀번호를 쓰지 않는다.
- 커밋 메시지: `[<브랜치 / 뒤 이름>] 한국어 제목` + 빈 줄 + `*` 불릿 + 마지막 줄 `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

```python
"""서버의 모든 시각은 KST(+09:00). 현재 시각은 now_kst()로만 만든다."""

from datetime import date, datetime, time
from zoneinfo import ZoneInfo

KST = ZoneInfo("Asia/Seoul")


def now_kst() -> datetime:
    return datetime.now(KST)


def start_of_day_kst(d: date) -> datetime:
    """그날 00:00 KST. 클래스 시작 날짜 → starts_at."""
    return datetime.combine(d, time.min, tzinfo=KST)


def add_years(dt: datetime, years: int) -> datetime:
    """dt에서 years년 뒤. 2월 29일은 다음 해에 없으므로 2월 28일로."""
    try:
        return dt.replace(year=dt.year + years)
    except ValueError:
        return dt.replace(year=dt.year + years, day=28)
```

```python
"""DB 컬럼 타입. 시각 컬럼은 모두 KstDateTime, Enum 컬럼은 enum_column()을 쓴다."""

from datetime import datetime
from enum import StrEnum

from sqlalchemy import DateTime
from sqlalchemy import Enum as SAEnum
from sqlalchemy.engine import Dialect
from sqlalchemy.types import TypeDecorator

from app.core.clock import KST


class KstDateTime(TypeDecorator[datetime]):
    """시간대가 있는 datetime만 받아 KST로 저장하고, 읽을 때 +09:00을 붙여 돌려준다.

    SQLite는 시간대를 저장하지 못하므로 KST 기준 시각을 시간대 없이 저장하고,
    읽을 때 KST를 다시 붙인다. PostgreSQL에서는 timestamptz로 저장된다.
    """

    impl = DateTime(timezone=True)
    cache_ok = True

    def process_bind_param(self, value: datetime | None, dialect: Dialect) -> datetime | None:
        if value is None:
            return None
        if value.tzinfo is None:
            raise ValueError("시간대 없는 datetime은 저장할 수 없습니다. now_kst()로 만든 값을 쓰세요.")
        kst = value.astimezone(KST)
        return kst.replace(tzinfo=None) if dialect.name == "sqlite" else kst

    def process_result_value(self, value: datetime | None, dialect: Dialect) -> datetime | None:
        if value is None:
            return None
        if value.tzinfo is None:
            return value.replace(tzinfo=KST)
        return value.astimezone(KST)


def enum_column(enum_cls: type[StrEnum]) -> SAEnum:
    """StrEnum을 대문자 문자열로 저장하는 컬럼 타입(DB 네이티브 enum은 쓰지 않음)."""
    return SAEnum(
        enum_cls,
        native_enum=False,
        length=20,
        values_callable=lambda e: [m.value for m in e],
        validate_strings=True,
    )
```

```python
"""비밀번호 저장: PBKDF2-HMAC-SHA512 (schema.md "비밀번호 저장"). 원문은 어디에도 남기지 않는다."""

import base64
import hashlib
import hmac
import secrets

ALGORITHM = "pbkdf2_sha512"
ITERATIONS = 200_000
SALT_BYTES = 16


def _derive(password: str, salt: bytes, iterations: int) -> bytes:
    return hashlib.pbkdf2_hmac("sha512", password.encode("utf-8"), salt, iterations)


def hash_password(password: str, *, iterations: int = ITERATIONS) -> str:
    salt = secrets.token_bytes(SALT_BYTES)
    digest = _derive(password, salt, iterations)
    salt_b64 = base64.b64encode(salt).decode("ascii")
    digest_b64 = base64.b64encode(digest).decode("ascii")
    return f"{ALGORITHM}${iterations}${salt_b64}${digest_b64}"


def verify_password(password: str, stored: str) -> bool:
    """저장된 반복 횟수로 다시 계산해 비교한다. 형식이 깨졌으면 False."""
    try:
        algorithm, iterations_text, salt_b64, digest_b64 = stored.split("$")
        iterations = int(iterations_text)
        salt = base64.b64decode(salt_b64, validate=True)
        expected = base64.b64decode(digest_b64, validate=True)
    except ValueError:  # 개수 불일치 · 숫자 아님 · base64 오류(binascii.Error는 ValueError 하위)
        return False
    if algorithm != ALGORITHM or iterations <= 0:
        return False
    return hmac.compare_digest(_derive(password, salt, iterations), expected)
```

```python
from collections.abc import Iterator
from typing import Any

from sqlalchemy import Engine, create_engine, event
from sqlalchemy.orm import DeclarativeBase, Session, sessionmaker
from sqlalchemy.pool import StaticPool

from app.core.config import get_settings


def create_db_engine(url: str) -> Engine:
    """SQLite면 스레드 검사를 끄고 외래 키 검사를 켠다. 메모리 DB(sqlite://)는 연결 하나를 공유한다."""
    if not url.startswith("sqlite"):
        return create_engine(url)
    kwargs: dict[str, Any] = {"connect_args": {"check_same_thread": False}}
    if url in ("sqlite://", "sqlite:///:memory:"):
        kwargs["poolclass"] = StaticPool
    engine = create_engine(url, **kwargs)

    @event.listens_for(engine, "connect")
    def _enable_foreign_keys(dbapi_connection: Any, _record: Any) -> None:
        cursor = dbapi_connection.cursor()
        cursor.execute("PRAGMA foreign_keys=ON")
        cursor.close()

    return engine


engine = create_db_engine(get_settings().database_url)
SessionLocal = sessionmaker(bind=engine, autoflush=False, expire_on_commit=False)


class Base(DeclarativeBase):
    """모든 ORM 모델의 부모. 모델은 app/models/ 에 둔다."""


def get_db() -> Iterator[Session]:
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()
```

```python
과제·문항 시드는 과제 테이블이 생기는 ④ 과제 API에서 추가한다.

from sqlalchemy import select
from sqlalchemy.orm import Session

def seed(db: Session, today: date | None = None) -> ClassRoom:

    now = now_kst()
    starts_at = start_of_day_kst(today or now.date())
    expires_at = add_years(starts_at, 1)
    plan = PLANS[PlanCode.SMALL]

    teacher = User(
        role=Role.TEACHER, email=SEED_TEACHER_EMAIL,
        password_hash=hash_password(SEED_TEACHER_PASSWORD), name="김교수",

    )
    db.add(klass)
    db.flush()

    db.add(ClassPayment(
        teacher_id=teacher.id, class_id=klass.id, plan=plan.code, student_limit=plan.student_limit,

        db.add(student)
        db.flush()
        db.add(Enrollment(
            class_id=klass.id, student_id=student.id, student_no=no,
            status=EnrollmentStatus.APPROVED, decided_at=now,
        ))
    db.commit()
    return klass
```
